Chapter 30: 프로젝트 구현: 코드로 만들어보자! (2)¶

🎯 이 장에서 배우는 것¶

  • 기본 기능 위에 추가 기능을 구현할 수 있다
  • 사용자가 쉽게 이해하도록 안내 메시지를 개선할 수 있다
  • 잘못된 입력에 대비한 예외 처리를 적용할 수 있다
  • 코드에 주석을 달아 정리하고 다른 사람도 읽을 수 있게 만들 수 있다

💡 왜 이걸 배우나요?¶

프로그램을 '만드는 것'과 '잘 만드는 것'은 다릅니다.

작동하는 코드는 기능이 구현되었다는 뜻이지만, 좋은 코드는 다른 사람(미래의 나 포함!)이 쉽게 이해하고 수정할 수 있다는 뜻입니다.

이 장에서는 핵심 기능이 완성된 프로젝트를 다듬어 완성도 있는 작품으로 만드는 방법을 배웁니다. 안내 메시지 개선, 예외 처리, 주석 추가—이 세 가지가 전문가 코드의 기초입니다. 🎨

📚 핵심 개념¶

개념 1: 예외 처리(Exception Handling)로 안정성 높이기¶

비유: 게임에서 "잘못된 선택지를 누르면 어떻게 되나요?"라고 묻는 것처럼, 프로그램도 예상 밖의 입력에 대비해야 합니다.

정의: 프로그램 실행 중 에러가 발생해도 멈추지 않고, 사용자에게 친절한 안내를 주는 기법입니다.

예시:

# ❌ 예외 처리 없음 (에러 발생하면 프로그램 멈춤)
age = int(input("나이: "))

# ✅ 예외 처리 있음 (안전함)
while True:
    try:
        age = int(input("나이(숫자만): "))
        if age < 0:
            print("나이는 0 이상이어야 합니다!")
            continue
        break
    except ValueError:
        print("⚠️ 숫자를 입력해주세요!")

try-except의 흐름:

graph LR A["코드 실행<br/>(try 블록)"] -->|정상| B["계속 실행"] A -->|에러 발생| C["except 블록<br/>(에러 처리)"] C --> D["안내 메시지 출력"]

개념 2: 사용자 경험(UX) 개선 - 명확한 안내 메시지¶

비유: 카페의 주문 방식이 명확하면 손님이 쉽게 주문하듯, 프로그램의 안내가 명확하면 사용자 실수가 줄어듭니다.

정의: 사용자가 다음에 무엇을 해야 하는지 즉시 알 수 있도록 메시지를 작성하는 것입니다.

개선 예시:

# ❌ 불친절
choice = input("입력: ")

# ✅ 친절
print("\n📋 메뉴를 선택하세요:")
print("1️⃣  추가")
print("2️⃣  삭제")
print("3️⃣  종료")
choice = input("\n선택(1-3): ")

개념 3: 주석(Comment) 작성 - 코드 읽기 쉽게 만들기¶

비유: 책의 주석(footnote)처럼, 코드도 설명이 있으면 복잡한 부분을 이해하기 쉽습니다.

정의: 코드가 무엇을 하는지, 왜 하는지 설명하는 텍스트입니다.

좋은 주석의 원칙:

# ✅ 좋은 주석: 이 코드가 왜 필요한지, 무엇을 하는지 설명
# 사용자가 음수를 입력하지 않도록 검증
if score < 0:
    print("점수는 0 이상이어야 합니다")

# ❌ 나쁜 주석: 코드 자체를 반복만 함
# score를 0과 비교
if score < 0:
    print("점수는 0 이상이어야 합니다")

# ❌ 너무 많은 주석: 코드가 복잡함을 의미
x = y + 1  # x에 y + 1을 더함
z = x * 2  # z에 x * 2를 곱함

개념 4: 함수로 코드 정리하기 - 재사용성 높이기¶

비유: 요리하기 전에 준비(mise en place)를 정리하듯, 반복되는 코드는 함수로 정리합니다.

정의: 같은 작업을 여러 번 하면 함수로 만들어 재사용하고, 코드를 짧고 읽기 쉽게 만듭니다.

예시:

# 반복되는 코드를 함수로 정리
def get_valid_number(prompt, min_val=0, max_val=None):
    """사용자로부터 유효한 정수 입력을 받는 함수"""
    while True:
        try:
            num = int(input(prompt))
            if num < min_val:
                print(f"⚠️ {min_val} 이상의 숫자를 입력하세요!")
                continue
            if max_val and num > max_val:
                print(f"⚠️ {max_val} 이하의 숫자를 입력하세요!")
                continue
            return num
        except ValueError:
            print("⚠️ 숫자를 입력해주세요!")

def get_menu_choice():
    """메뉴 선택을 받고 유효성 검사"""
    while True:
        try:
            choice = input("선택(1-4): ")
            if choice not in ["1", "2", "3", "4"]:
                print("⚠️ 1부터 4 사이의 숫자를 입력하세요!")
                continue
            return choice
        except KeyboardInterrupt:
            print("\n프로그램을 종료합니다.")
            exit()

def get_task_number(max_num):

    while True:
        try:
            num = int(input(f"번호(1-{max_num}): "))
            if num < 1 or num > max_num:
                print(f"⚠️ 1부터 {max_num} 사이의 숫자를 입력하세요!")

            return num

# 데이터 저장소
todo_list = []

def show_menu():

    print("\n" + "="*30)
    print("📝 할일 관리 프로그램")

def add_task():
    """새로운 할일을 추가"""
    task = input("➕ 추가할 할일: ").strip()
    if task:  # 빈 문자열 방지
        todo_list.append({"task": task, "done": False})
        print(f"✅ '{task}'가 추가되었습니다!")
    else:
        print("⚠️ 할일을 입력해주세요!")

def show_tasks():
    """할일 목록을 표시"""
    if not todo_list:
        print("\n📭 할일이 없습니다!")
        return

    print("\n📋 현재 할일:")
    for i, item in enumerate(todo_list, 1):
        status = "✅" if item["done"] else "⬜"
        print(f"  {i}. {status} {item['task']}")

def mark_done():
    """할일을 완료 표시"""
    show_tasks()
    if not todo_list:
        return

    while True:
        try:
            num = int(input("✔️ 완료할 할일 번호: "))
            if 1 <= num <= len(todo_list):
                todo_list[num - 1]["done"] = True
                print(f"✅ 완료되었습니다!")
                break
            else:
                print(f"⚠️ 1부터 {len(todo_list)} 사이의 숫자를 입력하세요!")
        except ValueError:
            print("⚠️ 숫자를 입력해주세요!")

def get_menu_choice():
    """사용자 선택을 받고 검증"""
    while True:
        choice = input("\n선택(1-4): ")
        if choice in ["1", "2", "3", "4"]:
            return choice
        print("⚠️ 1부터 4 사이의 숫자를 입력하세요!")

def main():
    """프로그램의 메인 루프"""
    print("🎉 할일 관리 프로그램을 시작합니다!")

    while True:
        show_menu()
        choice = get_menu_choice()

        if choice == "1":
            add_task()
        elif choice == "2":
            show_tasks()
        elif choice == "3":
            mark_done()
        elif choice == "4":
            print("\n👋 프로그램을 종료합니다. 수고하셨어요!")
            break

# 프로그램 시작
if __name__ == "__main__":
    main()

import json

def save_tasks(filename="tasks.json"):
    """할일 목록을 파일에 저장"""
    try:
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(todo_list, f, ensure_ascii=False, indent=2)
        print("💾 저장되었습니다!")
    except Exception as e:
        print(f"⚠️ 저장 실패: {e}")

def load_tasks(filename="tasks.json"):
    """파일에서 할일 목록을 불러오기"""
    global todo_list
    try:
        with open(filename, "r", encoding="utf-8") as f:
            todo_list = json.load(f)
        print("📂 데이터가 로드되었습니다!")
    except FileNotFoundError:
        print("ℹ️ 저장된 데이터가 없습니다. 새로 시작합니다.")
    except Exception as e:
        print(f"⚠️ 로드 실패: {e}")

    print("4️⃣  할일 삭제하기")
    print("5️⃣  프로그램 종료하기")
    print("="*35)

def get_menu_choice():
    """유효한 메뉴 선택 받기"""
    while True:
        choice = input("\n선택(1-5): ").strip()
        if choice in ["1", "2", "3", "4", "5"]:
            return choice
        print("⚠️  1부터 5 사이의 숫자를 입력하세요!")

def add_task():
    """새 할일 추가"""
    task = input("\n➕ 추가할 할일: ").strip()
    if task:
        todo_list.append({"task": task, "done": False})
        print(f"✅ '{task}'가 추가되었습니다!")
    else:
        print("⚠️  할일을 입력해주세요!")

def show_tasks():
    """할일 목록 표시"""
    if not todo_list:
        print("\n📭 할일이 없습니다!")
        return

    print("\n📋 현재 할일:")
    for i, item in enumerate(todo_list, 1):
        status = "✅" if item["done"] else "⬜"
        done_mark = " (완료됨)" if item["done"] else ""
        print(f"  {i}. {status} {item['task']}{done_mark}")

def mark_done():
    """할일을 완료로 표시"""
    show_tasks()
    if not todo_list:
        return

    while True:
        try:
            num = int(input("\n✔️  완료할 할일 번호: "))
            if 1 <= num <= len(todo_list):
                todo_list[num - 1]["done"] = True
                print(f"✅ '{todo_list[num - 1]['task']}'이(가) 완료되었습니다!")
                break
            else:
                print(f"⚠️  1부터 {len(todo_list)} 사이의 숫자를 입력하세요!")
        except ValueError:
            print("⚠️  숫자를 입력해주세요!")

def delete_task():
    """할일 삭제"""
    show_tasks()
    if not todo_list:
        return

    while True:
        try:
            num = int(input("\n🗑️  삭제할 할일 번호: "))
            if 1 <= num <= len(todo_list):
                removed = todo_list.pop(num - 1)
                print(f"✅ '{removed['task']}'이(가) 삭제되었습니다!")
                break
            else:
                print(f"⚠️  1부터 {len(todo_list)} 사이의 숫자를 입력하세요!")
        except ValueError:
            print("⚠️  숫자를 입력해주세요!")

def save_tasks():
    """할일 목록을 파일에 저장"""
    try:
        with open(FILENAME, "w", encoding="utf-8") as f:
            json.dump(todo_list, f, ensure_ascii=False, indent=2)
        print("💾 저장되었습니다!")
    except Exception as e:
        print(f"⚠️  저장 실패: {e}")

def load_tasks():
    """파일에서 할일 목록 불러오기"""
    global todo_list
    try:
        with open(FILENAME, "r", encoding="utf-8") as f:
            todo_list = json.load(f)
        print("📂 이전 데이터가 로드되었습니다!")
    except FileNotFoundError:
        print("ℹ️  저장된 데이터가 없습니다. 새로 시작합니다.")
    except Exception as e:
        print(f"⚠️  로드 실패: {e}")

def main():
    """메인 프로그램 루프"""
    print("\n🎉 할일 관리 프로그램을 시작합니다!")
    load_tasks()  # 시작할 때 이전 데이터 불러오기

    while True:
        show_menu()
        choice = get_menu_choice()

        if choice == "1":
            add_task()
        elif choice == "2":
            show_tasks()
        elif choice == "3":
            mark_done()
        elif choice == "4":
            delete_task()
        elif choice == "5":
            save_tasks()  # 종료 전에 데이터 저장
            print("\n👋 프로그램을 종료합니다. 수고하셨어요!")
            break

# 프로그램 실행 진입점
if __name__ == "__main__":
    main()

def search_tasks():
    """키워드로 할일 검색"""
    keyword = input("🔍 검색할 키워드: ").strip().lower()
    results = [item for item in todo_list 
               if keyword in item["task"].lower()]

    if results:
        print(f"\n📋 '{keyword}'에 대한 검색 결과:")
        for i, item in enumerate(results, 1):
            status = "✅" if item["done"] else "⬜"
            print(f"  {i}. {status} {item['task']}")
    else:
        print(f"⚠️ '{keyword}'에 대한 할일이 없습니다!")
