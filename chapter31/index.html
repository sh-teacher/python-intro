Chapter 31: 발표와 공유: 내가 만든 작품을 보여줘!¶

🎯 이 장에서 배우는 것¶

  • 프로젝트를 청중 앞에서 효과적으로 발표할 수 있다
  • 프로그램 시연과 코드 설명을 구조화할 수 있다

💡 왜 이걸 배우나요?¶

프로그래밍은 혼자만의 활동이 아닙니다. 여러분이 만든 프로그램을 다른 사람에게 보여주고 설명하는 능력은:

🎤 취업면접에서 포트폴리오를 설명할 때 💼 직장에서 팀과 협업할 때 🤝 오픈소스 커뮤니티에서 코드를 공유할 때 📈 피드백을 받아 코드를 개선할 때

...모두 필수적인 기술입니다. 발표와 공유는 단순한 보여주기가 아니라, 자신의 코드를 객관적으로 보는 능력을 키우는 훈련입니다.

📚 핵심 개념¶

개념 1: 발표 3단계 구조¶

비유: 영상을 찍을 때 도입-전개-결말이 있듯이, 발표도 시작-중간-끝이 있어야 청중이 따라갈 수 있습니다.

구조: - 문제 제시 (30초): "왜 이걸 만들었나?" - 시연과 설명 (2-3분): "어떻게 작동하는가?" - 배운 점 (30초): "무엇을 얻었나?"

# 발표 스크립트 작성 예제
presentation = {
    "문제정의": "여행 일정 관리가 복잡했어요",
    "솔루션": "여행 계획 자동 정렬 프로그램 만들었어요",
    "주요기능": [
        "장소별 거리 계산",
        "가장 짧은 경로 찾기",
        "일정 시간 예측"
    ],
    "배운점": [
        "데이터 구조의 중요성",
        "알고리즘적 사고",
        "사용자 경험 고려"
    ]
}

print(f"문제: {presentation['문제정의']}")
print(f"해결책: {presentation['솔루션']}")

개념 2: 효과적인 시연 준비¶

정의: 프로그램이 제대로 작동하도록 미리 테스트하고, 실패할 경우의 대비책을 마련하는 것입니다.

체크리스트: - ✅ 필요한 파일/데이터가 모두 준비되었나? - ✅ 입력값 예제를 여러 개 준비했나? - ✅ 실패 사례도 시연할 건가? - ✅ 프로그램이 예상대로 실행되나? - ✅ 프로젝터/스크린 해상도는 괜찮나?

개념 3: 건설적 피드백¶

비유: 친구의 요리를 평가할 때 "맛없어"라고만 하는 것보다, "소금이 좀 더 필요할 것 같아. 다음엔 이렇게 해봐"라고 하는 게 더 도움이 됩니다.

좋은 피드백의 3요소: - 🎯 구체성: "좋았어"가 아니라 "UI가 직관적이라 좋았어" - 💡 제안성: 문제만 지적하지 말고 개선 방안 제시 - 🤝 존중: 상대의 노력을 먼저 인정하고 시작

개념 4: 학습 회고와 성장 인식¶

정의: 지난 31강의 학습을 돌아보며, 자신이 처음엔 할 수 없었던 것들을 이제 할 수 있게 된 변화를 기록하는 과정입니다.

# 학습 회고 자가 진단
learning_retrospective = {
    "처음": {
        "할수있는것": ["변수 할당", "print() 함수"],
        "못하는것": ["데이터 구조", "파일 입출력", "GUI 프로그램"]
    },
    "지금": {
        "할수있는것": [
            "변수 할당",
            "print() 함수",
            "리스트/딕셔너리 다루기",
            "반복문과 조건문",
            "함수 정의",
            "클래스와 객체",
            "파일 입출력",
            "간단한 GUI 만들기",
            "API 사용",
            "데이터 분석"
        ],
        "실제프로젝트": ["여행계획앱", "채팅봇", "게임"]
    },
    "다음": {
        "도전해볼것": ["웹개발", "데이터과학", "모바일앱", "AI/ML"]
    }
}

def print_growth():
    before = len(learning_retrospective["처음"]["할수있는것"])
    after = len(learning_retrospective["지금"]["할수있는것"])
    growth = after - before

    print(f"📊 31강을 거친 성장")
    print(f"처음: {before}가지 → 지금: {after}가지 ({growth}가지 증가!)")
    print(f"실제 프로젝트: {len(learning_retrospective['지금']['실제프로젝트'])}개 완성")

print_growth()

Step 1: 발표 스크립트 작성하기¶

프로젝트 발표를 위한 3분짜리 스크립트를 작성합니다. 실제 프로젝트를 기반으로 진행하세요.

# 발표 스크립트 작성 도구
class PresentationScript:
    def __init__(self, project_name):
        self.project_name = project_name
        self.script = {}

    def add_introduction(self, problem, motivation):
        self.script["도입"] = {
            "문제": problem,
            "동기": motivation,
            "지속시간": "30초"
        }
        return self

    def add_demonstration(self, features):
        self.script["시연"] = {
            "주요기능": features,
            "예상시간": "2분"
        }
        return self

    def add_learning(self, lessons):
        self.script["배움"] = {
            "배운점": lessons,
            "지속시간": "30초"
        }
        return self

    def print_script(self):
        print(f"\n📋 [{self.project_name}] 발표 스크립트\n")

        print("🎤 도입 (30초)")
        intro = self.script.get("도입", {})
        print(f"  문제: {intro.get('문제', 'N/A')}")
        print(f"  동기: {intro.get('동기', 'N/A')}")

        print("\n🎬 시연 (2분)")
        demo = self.script.get("시연", {})
        for i, feature in enumerate(demo.get("주요기능", []), 1):
            print(f"  {i}. {feature}")

        print("\n💡 배움 (30초)")
        learning = self.script.get("배움", {})
        for i, lesson in enumerate(learning.get("배운점", []), 1):
            print(f"  {i}. {lesson}")

# 사용 예제
script = (PresentationScript("여행 계획 앱")
    .add_introduction(
        problem="여행 일정이 복잡해서 관리하기 어려웠어요",
        motivation="Python으로 자동 정렬 프로그램을 만들고 싶었어요"
    )
    .add_demonstration([
        "여행지 입력하면 최적 경로 추천",

Step 2: 실행 환경 미리 테스트하기¶

발표 전에 프로그램이 제대로 작동하는지 점검합니다.

# 시연 테스트 체크리스트
class DemoTest:
    def __init__(self, program_name):
        self.program_name = program_name
        self.checklist = {
            "파일준비": False,
            "데이터로드": False,
            "기본입력": False,
            "예외처리": False,
            "화면출력": False
        }

    def test_file_existence(self, required_files):
        print(f"✓ 파일 확인: {required_files}")
        self.checklist["파일준비"] = True

    def test_data_loading(self, data_sample):
        print(f"✓ 데이터 로드: {len(data_sample)}개 항목")
        self.checklist["데이터로드"] = True

    def test_basic_input(self, test_cases):
        print(f"✓ 기본 입력 테스트: {len(test_cases)}개 케이스 통과")
        self.checklist["기본입력"] = True

    def test_error_handling(self):
        print("✓ 잘못된 입력에 대한 오류 처리 확인")
        self.checklist["예외처리"] = True

    def test_output(self):
        print("✓ 출력 화면 확인 (글씨 크기, 가독성)")
        self.checklist["화면출력"] = True

    def report(self):
        passed = sum(self.checklist.values())
        total = len(self.checklist)
        print(f"\n📊 시연 준비 완료: {passed}/{total}")
        if passed == total:
            print("🎉 발표 준비 완료!")
        else:
            print("⚠️ 아직 준비할 것이 있습니다")

# 사용 예
tester = DemoTest("채팅 봇")
tester.test_file_existence(["chatbot.py", "responses.txt"])
tester.test_data_loading(["안녕", "반갑습니다", "어떻게 지내"])
tester.test_basic_input([("안녕하세요", "expected_response")])
tester.test_error_handling()
tester.test_output()
tester.report()

# 피드백 수집 및 정리
class FeedbackCollector:
    def __init__(self):
        self.feedbacks = []

    def add_feedback(self, reviewer_name, positive, improvement, question):
        feedback = {
            "작성자": reviewer_name,
            "좋았던점": positive,
            "개선점": improvement,
            "질문": question
        }
        self.feedbacks.append(feedback)

    def show_summary(self):
        print("\n📋 피드백 요약\n")

        if not self.feedbacks:
            print("아직 피드백이 없습니다")
            return

        print(f"총 {len(self.feedbacks)}명의 피드백\n")

        all_positive = []
        all_improvements = []

        for fb in self.feedbacks:
            print(f"🧑 {fb['작성자']}님 피드백:")
            print(f"  좋았던 점: {fb['좋았던점']}")
            print(f"  개선점: {fb['개선점']}")
            print(f"  질문: {fb['질문']}\n")

            all_positive.append(fb['좋았던점'])
            all_improvements.append(fb['개선점'])

        print("📊 자주 나온 개선 점:")
        improvement_counts = {}
        for imp in all_improvements:
            improvement_counts[imp] = improvement_counts.get(imp, 0) + 1

        for imp, count in sorted(improvement_counts.items(), 
                                 key=lambda x: x[1], reverse=True):
            print(f"  • {imp} ({count}명 제안)")

# 학습 회고록 양식
class LearningRetrospective:
    def __init__(self, student_name):
        self.name = student_name
        self.content = {}

    def add_journey(self, before, after):
        self.content["변화"] = f"{before} → {after}"

    def add_challenge(self, challenge, solution):
        self.content["도전과제"] = {
            "문제": challenge,
            "해결방법": solution
        }

    def add_lessons(self, lessons):
        self.content["배운점"] = lessons

    def add_future(self, goals):
        self.content["앞으로의목표"] = goals

    def print_retrospective(self):
        print(f"\n📖 {self.name}의 파이썬 학습 회고\n")

        if "변화" in self.content:
            print(f"변화: {self.content['변화']}\n")

        if "도전과제" in self.content:
            challenge = self.content["도전과제"]
            print(f"가장 어려웠던 부분: {challenge['문제']}")
            print(f"해결 방법: {challenge['해결방법']}\n")

        if "배운점" in self.content:
            print("배운 점:")
            for i, lesson in enumerate(self.content["배운점"], 1):
                print(f"  {i}. {lesson}")
            print()

        if "앞으로의목표" in self.content:
            print("앞으로의 목표:")
            for goal in self.content["앞으로의목표"]:
                print(f"  🎯 {goal}")

# 사용 예
retrospective = LearningRetrospective("박준호")
retrospective.add_journey(
    "print()만 알던 초보자",

)
retrospective.add_lessons([

    "문제를 단계별로 나누어 풀 수 있게 되었어요",
    "프로그래밍으로 실제 문제를 해결할 수 있다는 걸 알았어요"
])

    "웹개발 배우기",

# 학급 프로젝트 평가 시스템
class ProjectEvaluation:
    def __init__(self, evaluator_name):
        self.evaluator = evaluator_name
        self.evaluations = []

    def evaluate_project(self, project_name, 
                         creativity, functionality, presentation):
        """
        creativity, functionality, presentation: 1-5점
        """
        evaluation = {
            "프로젝트명": project_name,
            "창의성": creativity,
            "기능성": functionality,
            "발표력": presentation,
            "평균점수": (creativity + functionality + presentation) / 3
        }
        self.evaluations.append(evaluation)

    def show_results(self):
        print(f"\n🎓 {self.evaluator}님의 평가\n")

        for eval in self.evaluations:
            print(f"📌 {eval['프로젝트명']}")
            print(f"   창의성: {'⭐' * eval['창의성']} ({eval['창의성']}/5)")
            print(f"   기능성: {'⭐' * eval['기능성']} ({eval['기능성']}/5)")
            print(f"   발표력: {'⭐' * eval['발표력']} ({eval['발표력']}/5)")
            print(f"   평균: {eval['평균점수']:.1f}/5\n")

# 사용 예
evaluator = ProjectEvaluation("오예진")
evaluator.evaluate_project("여행 계획 앱", 5, 4, 5)
evaluator.evaluate_project("채팅 봇", 4, 4, 3)
evaluator.evaluate_project("게임", 5, 5, 4)
evaluator.show_results()

# 통합 발표 및 공유 시스템

class PresentationManager:
    """프로젝트 발표와 피드백을 관리하는 통합 시스템"""

    def __init__(self, event_name):
        self.event_name = event_name
        self.projects = []
        self.feedbacks = {}

    def register_project(self, project_name, presenter_name):
        project = {
            "이름": project_name,
            "발표자": presenter_name,
            "피드백": []
        }
        self.projects.append(project)
        self.feedbacks[project_name] = []
        return project

    def add_feedback(self, project_name, reviewer, comment):
        self.feedbacks[project_name].append({
            "작성자": reviewer,
            "내용": comment
        })

    def show_event_summary(self):
        print(f"\n🎉 {self.event_name}\n")
        print(f"참여 프로젝트: {len(self.projects)}개\n")

        for i, project in enumerate(self.projects, 1):
            print(f"{i}. {project['이름']} - {project['발표자']}님")
            feedback_count = len(self.feedbacks[project['이름']])
            print(f"   받은 피드백: {feedback_count}개\n")

# 실행
